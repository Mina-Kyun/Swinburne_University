// ===== MATCHSTICKS GAME - ABI VERSION =====
//---- The Main Program ----
Main_Loop:
      BL Name           ; call Name()
      BL Number         ; call Number()
      BL Start_Matchsticks ; init display
Main_Game_Loop:
      BL Draw_Matchsticks ; draw graphics
      BL Player         ; player turn
      CMP R0, #0        ; check = 0
      BEQ Draw          ; jump draw
      CMP R0, #1        ; check = 1
      BEQ Player_Win    ; Player Win
      BL Computer_Turn  ; computer turn
      CMP R0, #0        ; check = 0
      BEQ Draw
      CMP R0, #1
      BEQ Player_Lose   ; Player Lose
      B Main_Game_Loop  ; loop game
Restart_Game:
      BL Number         ; re-input number
      BL Start_Matchsticks ; reset game
      B Main_Game_Loop
// ==== STAGE 1: GAME SETUP ====
// ---- Ask Name ----
Name:
      PUSH {LR}         ; save return addr
      MOV R1, #name_buffer ; addr (store name)
      MOV R3, #msg_ask_name ; string addr
      STR R3, .WriteString ; print string
      STR R1, .ReadString ; input string → buffer
      POP {LR}          ; restore LR
      RET               ; return
// ---- Ask Matchsticks (10–100) ----
Number:
      PUSH {LR}
Loop_Input:
      MOV R3, #msg_player ; load "Player"
      STR R3, .WriteString ; print
      STR R1, .WriteString ; print name
      MOV R3, #msg_ask_number ; load question
      STR R3, .WriteString ; print
      LDR R0, .InputNum ; read number → R0
      CMP R0, #10       ; compare R0 vs 10
      BLT Loop_Input    ; if < → retry
      CMP R0, #100      ; compare vs 100
      BGT Loop_Input    ; if > → retry
      POP {LR}
      RET 
//==== STAGE 2: SINGLE PLAYER INPUT ====
// ---- GAME LOOP ----
Start_Matchsticks:
      PUSH {LR}
      BL Draw_Matchsticks ; draw first time
      MOV R3, #msg_player
      STR R3, .WriteString
      STR R1, .WriteString
      MOV R3, #msg_remaining_1
      STR R3, .WriteString
      STR R0, .WriteUnsignedNum ; print number
      MOV R3, #msg_remaining_2
      STR R3, .WriteString
      POP {LR}
      RET
// ---- PLAYER TURN ----
Player:
      PUSH {LR}
Player_Turn:
      MOV R3, #msg_player
      STR R3, .WriteString
      STR R1, .WriteString
      MOV R3, #msg_ask_remove
      STR R3, .WriteString ; ask input
      LDR R4, .InputNum ; read input → R4
      CMP R4, #1
      BLT Player_Turn   ; <1 → retry
      CMP R4, #7
      BGT Player_Turn   ; >7 → retry
      CMP R4, R0
      BGT Player_Turn   ; >remaining → retry
      SUB R0, R0, R4    ; update matchsticks
      MOV R2, #0
      STR R2, .ClearScreen ; clear screen
      BL Draw_Matchsticks ; redraw
      POP {LR}
      RET 
// ==== STAGE 3: IMPLEMENT HUMAN VERSUS COMPUTER ====
// ---- COMPUTER TURN ----
Computer_Turn:
      PUSH {LR}
      MOV R3, #msg_computer_turn
      STR R3, .WriteString ; print message
Computer_Loop:
      LDR R5, .Random   ; random value
      LSR R5, R5, #29   ; shrink range (0–7)
      CMP R5, #0
      BEQ Computer_Loop ; avoid 0
      CMP R5, R0
      BGT Computer_Loop ; avoid invalid
      SUB R0, R0, R5    ; update total
      MOV R2, #0
      STR R2, .ClearScreen 
      BL Draw_Matchsticks
      MOV R3, #msg_remaining
      STR R3, .WriteString
      STR R0, .WriteUnsignedNum
      POP {LR}
      RET
// ---- RESULT ----
Player_Win:
      STR R1, .WriteString 
      MOV R3, #you_win
      STR R3, .WriteString
      B Game_Over
Player_Lose:
      STR R1, .WriteString
      MOV R3, #you_lose
      STR R3, .WriteString
      B Game_Over
Draw:
      MOV R3, #msg_draw
      STR R3, .WriteString
      B Game_Over
// ---- GAME OVER ----
Game_Over:
      MOV R3, #msg_game_over
      STR R3, .WriteString ; print end
      B Replay
// ---- REPLAY ----
Replay:
      MOV R3, #msg_ask_for_replay
      STR R3, .WriteString ; ask replay
      MOV R0, #choice
      STR R0, .ReadString ; input
      LDRB R0, [R0]     ; load 1 char
      CMP R0, #121      ; 'y'
      BEQ Change_Name
      CMP R0, #110      ; 'n'
      BEQ End
      B Replay          ; invalid → retry
// ---- CHANGE NAME ----
Change_Name:
      STR R1, .WriteString
      MOV R3, #msg_change_name
      STR R3, .WriteString
      MOV R0, #choice
      STR R0, .ReadString
      LDRB R0, [R0]
      CMP R0, #121      ; 'y'
      BEQ Main_Loop     ; change name
      CMP R0, #110      ; 'n'
      BEQ Restart_Game  ; keep name
      B Change_Name
End:
      HALT              ; stop program
// ===== STAGE 4: GRAPHICS =====
Draw_Matchsticks:
      PUSH {R2, R4, R5, R6, R7, R8, R9, LR} ; save registers used in this function
      MOV R7, R0        ; total number of matchsticks
      MOV R6, #0        ; number of matchsticks drawn
      MOV R8, #1        ; X coordinate (column)
      MOV R9, #1        ; Y coordinate (row)
Draw_Loop:
      CMP R6, R7        ; check if all matchsticks are drawn
      BEQ Done_Draw     ; if yes → exit
      MOV R2, #.PixelScreen ; base address of screen
      LSL R4, R8, #2    ; X * 4 (each pixel = 4 bytes)
      LSL R5, R9, #8    ; Y * 256 (each row = 256 bytes)
      ADD R5, R5, R4    ; pixel offset
; ===== HEAD (red tip) =====
      MOV R3, #.red     ; set color = red
      STR R3, [R2, R5]  ; draw head pixel
; ===== BODY (stick) =====
      MOV R3, #.orange  ; set color = yellow
      ADD R5, R5, #4    ; move right (next pixel)
      STR R3, [R2, R5]  ; draw body pixel 1
      ADD R5, R5, #4    ; move right again
      STR R3, [R2, R5]  ; draw body pixel 2
      ADD R5, R5, #4
      STR R3, [R2, R5]
      ADD R5, R5, #4
      STR R3, [R2, R5]
; ===== MOVE POSITION =====
      ADD R8, R8, #6    ; X += 6 (move right)
      ADD R6, R6, #1    ; increase drawn count
      CMP R8, #60       ; check end of row
      BLT Draw_Loop     ; if not → continue same row
; ===== NEW LINE =====
      MOV R8, #1        ; reset X to start
      ADD R9, R9, #3    ; Y += 3 (move down)
      CMP R9, #47       ; check screen limit
      BLT Draw_Loop     ; if not → continue drawing
Done_Draw:
      POP {R2, R4, R5, R6, R7, R8, R9, LR} ; restore registers
      RET               ; return
//

// ==== DATA ====
//---- Stage 1 ----
msg_ask_name: .ASCIZ "\nEnter your name:"
msg_ask_number: .ASCIZ ", Enter number of matchsticks (10-100):"
msg_player: .ASCIZ "\nPlayer: "
msg_matchsticks: .ASCIZ ", Matchsticks:"
// ---- Stage 2 ----
msg_remaining_1: .asciz ", There are"
msg_remaining_2: .asciz "matchsticks remaining."
msg_ask_remove: .asciz ", How many do you want to remove (1-7)?\n"
msg_remaining: .asciz "\nRemaining: "
msg_game_over: .asciz "\nGame Over!"
// ---- Stage 3 ----
msg_computer_turn: .asciz "\nComputer's turn\n"
you_win: .asciz ",YOU WIN!"
you_lose: .asciz ",YOU LOSE!"
msg_draw: .asciz "\nIt's a DRAW!"
msg_ask_for_replay: .asciz "\nPlay again (y/n)?\n"
// 
msg_change_name: .asciz ", Do you want to change your name (y/n)?"
//
name_buffer: .block 32
choice: .block 4
